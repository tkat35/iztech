  1. Pil en son takılır, ilk çıkarılır. Kablo değiştirirken pil çıkık olsun.
  2. Kod yüklerken tekerlekler havada. Robotu bir kutunun üstüne koyun.

  4. Keski ve pense: kabloyu kendinizden uzağa doğru kesin; kesilen uçları çöpe atın.
  5. Robot yerde çalışır, masada değil. Geri giden kodlar robotu masadan düşürebilir.
  6. USB kablosu takılıyken robotu sürmeyin. Kodu yükleyin, kabloyu çıkarın, sonra RESET'e basın.

// Acildiktan 3 sn sonra bir kez calisir. Tekrar icin RESET dugmesine basin.

const byte SOL_PWM = 5, SOL_IN1 = 7, SOL_IN2 = 8;
const byte SAG_PWM = 6, SAG_IN1 = 9, SAG_IN2 = 10;

const int HIZ = 120;        // 0-255 arasi
const int SAG_AYAR = 0;     // Robot sola kayiyorsa eksi (-10), saga kayiyorsa arti (+10)
const int ILERI_MS = 1500;  // Ileri gitme suresi (ms)
const int DONUS_HIZI = 100; // Donus hizi. Diger kodlarda da ayni deger kullanilir.
const int DONUS_MS = 450;   // Tam 90 derece donene kadar bu sayiyi degistirin

void motor(byte pwm, byte in1, byte in2, int hiz) {
  hiz = constrain(hiz, -255, 255);
  digitalWrite(in1, hiz >= 0 ? HIGH : LOW);
  digitalWrite(in2, hiz >= 0 ? LOW : HIGH);
  analogWrite(pwm, abs(hiz));
}

void sur(int sol, int sag) {
  motor(SOL_PWM, SOL_IN1, SOL_IN2, sol);
  motor(SAG_PWM, SAG_IN1, SAG_IN2, sag);
}

void setup() {
  pinMode(SOL_PWM, OUTPUT); pinMode(SOL_IN1, OUTPUT); pinMode(SOL_IN2, OUTPUT);
  pinMode(SAG_PWM, OUTPUT); pinMode(SAG_IN1, OUTPUT); pinMode(SAG_IN2, OUTPUT);
  sur(0, 0);
  delay(3000);                                  // Robotu yere koymak icin zaman

  sur(0, 0);                 delay(500);
  sur(DONUS_HIZI, -DONUS_HIZI); delay(DONUS_MS); // Yerinde SAGA don: robot kendi sagina donmeli!
  sur(0, 0);                 delay(500);
  sur(HIZ, HIZ + SAG_AYAR);  delay(ILERI_MS);   // Yine ileri
  sur(0, 0);
}

Bir adım öte: sensörü yerden 0,5 cm, 1 cm, 1,5 cm yukarıda deneyin. Hangi yükseklik en iyi okuyor? Unutmayın: yarışmada pistin ışığı farklı olacak ve itiraz kabul edilmiyor; ayarı orada yeniden yapacaksınız.

Kodu göster
İndir
// 2. CIZGI SENSORU: dort gozu oku, cizginin nerede oldugunu yaz
// OUT1-OUT4 soldan saga A0-A3. Seri Monitor: 9600 baud.

const byte CIZGI[4] = {A0, A1, A2, A3};
const bool SIYAH_LOW = true;  // Siyahta 0 okunuyorsa true, 1 okunuyorsa false yapin

bool cizgiVar(byte pin) {
  int d = digitalRead(pin);
  return SIYAH_LOW ? d == LOW : d == HIGH;
}

void setup() {
  Serial.begin(9600);
  for (byte i = 0; i < 4; i++) pinMode(CIZGI[i], INPUT);
}

void loop() {
  bool s[4];
  for (byte i = 0; i < 4; i++) {
    s[i] = cizgiVar(CIZGI[i]);
    Serial.print(s[i] ? "# " : ". ");   // # = siyah, . = beyaz
  }

  int sol = s[0] + s[1];
  int sag = s[2] + s[3];
  Serial.print("  -> ");
  if (sol + sag == 0)      Serial.println("cizgi yok");
  else if (sol + sag == 4) Serial.println("genis bant ya da kavsak");
  else if (sol > sag)      Serial.println("cizgi solda");
  else if (sag > sol)      Serial.println("cizgi sagda");
  else                     Serial.println("cizgi ortada");

  delay(150);
}

Kodu göster
İndir
// 3. MESAFE SENSORU (HC-SR04 / HY-SRF05): olc, ortalamasini al, cubuk ciz
// TRIG -> 11, ECHO -> 12. Seri Monitor: 9600 baud.

const byte TRIG = 11, ECHO = 12;

long olc() {
  digitalWrite(TRIG, LOW);  delayMicroseconds(2);
  digitalWrite(TRIG, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG, LOW);
  unsigned long sure = pulseIn(ECHO, HIGH, 25000);  // En fazla 25 ms bekle
  return sure == 0 ? 0 : sure / 58;                 // 0 = olcum yok
}

// 5 olcumun ortalamasi. Olcum yoksa (0) ortalamaya katilmaz.
long mesafeCm() {
  long toplam = 0;
  int n = 0;
  for (int i = 0; i < 5; i++) {
    long cm = olc();
    if (cm > 0) { toplam += cm; n++; }
    delay(10);
  }
  return n > 0 ? toplam / n : 0;
}

void setup() {
  Serial.begin(9600);
  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);
}

void loop() {
  long cm = mesafeCm();
  if (cm == 0) {
    Serial.println("olcum yok (cok uzak ya da egik yuzey)");
  } else {
    Serial.print(cm);
    Serial.print(" cm  ");
    long cubuk = cm > 60 ? 30 : cm / 2;   // Her = isareti 2 cm
    for (long i = 0; i < cubuk; i++) Serial.print('=');
    Serial.println('|');
  }
  delay(200);
}

// 4. SERVO: Seri Monitor'den aci yaz, servo o aciya gitsin
// Sinyal -> 3. Seri Monitor: 9600 baud, satir sonu "Yeni satir".

#include <Servo.h>

Servo kol;
const byte SERVO_PIN = 3;
const int EN_AZ = 10, EN_COK = 170;   // Guvenli sinirlar; mekanizmaya gore daraltin

void setup() {
  Serial.begin(9600);
  Serial.setTimeout(50);
  kol.attach(SERVO_PIN);
  kol.write(90);
  Serial.println("Bir aci yazip Enter'a basin (10-170):");
}

void loop() {
  if (Serial.available()) {
    int aci = Serial.parseInt();
    while (Serial.available()) Serial.read();   // Satir sonunu temizle

    if (aci >= EN_AZ && aci <= EN_COK) {
      kol.write(aci);
      Serial.print("Servo: ");
      Serial.println(aci);
    } else {
      Serial.println("10 ile 170 arasinda bir sayi yazin.");
    }
  }
}

// 5. MESAFE + MOTOR: elimi takip et
// Robot onundeki ele ya da kutuya 15 cm uzaklikta durmaya calisir:
// uzaklasirsan gelir, yaklasirsan geri kacar.

const byte SAG_PWM = 6, SAG_IN1 = 9, SAG_IN2 = 10;
const byte TRIG = 11, ECHO = 12;

const int HIZ = 90;
const int HEDEF_CM = 15;
const int TOLERANS = 3;     // 12-18 cm arasi "tamam" sayilir
const int MENZIL_CM = 60;   // Bundan uzaksa kimse yok, bekle

void motor(byte pwm, byte in1, byte in2, int hiz) {
  hiz = constrain(hiz, -255, 255);
  digitalWrite(in1, hiz >= 0 ? HIGH : LOW);
  digitalWrite(in2, hiz >= 0 ? LOW : HIGH);
  analogWrite(pwm, abs(hiz));
}
void sur(int sol, int sag) {
  motor(SOL_PWM, SOL_IN1, SOL_IN2, sol);
  motor(SAG_PWM, SAG_IN1, SAG_IN2, sag);
}

long olc() {
  digitalWrite(TRIG, LOW);  delayMicroseconds(2);
  digitalWrite(TRIG, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG, LOW);

  delay(30);
}

      break;
    }

    case YAKLAS: {                    // Yavasca nesneye yaklas
      long cm = mesafeCm();
      if (cm == 0) dur();             // Olcum yoksa ilerleme
      else if (cm > YAKIN_CM) sur(60, 60);
      else { dur(); adimaGec(AL); }
      break;
    }

    case AL:                          // Kolu kapat, biraz geri cekil
      kol.write(SERVO_KAPALI);
      if (adimSuresi() > 600) sur(-60, -60);
      if (adimSuresi() > 1000) { dur(); adimaGec(DONUS); }
      break;

    case DONUS:                       // Yerinde 180 derece don
      sur(DONUS_HIZI, -DONUS_HIZI);
      if (adimSuresi() > (unsigned long)DONUS_MS * 2) { dur(); adimaGec(BIRAK); }
      break;

    case BIRAK:                       // Hedef: 4 sensor de siyah (genis bant)
      if (gorenSensor == 4) {
        dur();
        kol.write(SERVO_ACIK);
        adimaGec(BITTI);
      } else {
        cizgiTakip(70);
      }
      break;

    case BITTI:
      dur();
      break;
  }
  delay(10);
}
Pinler örnektir. Yarışmada verilecek elektronik parçaların modeli kurallarda yazmıyor. Farklı bir sürücü ya da sensör çıkarsa yalnız ilgili modülü değiştirin.
Motoru Arduino'nun 5V pininden beslemeyin. Motorlar sürücüden ve pilden beslenir; tüm GND'ler ortak olmalı.
Her çalışan sürümü kaydedin. Örnek: tyy_v3_kiskac_calisiyor.ino. Bir şey bozulursa bir önceki sürüme dönersiniz.

Yarışma günü: 24 saat planı

Saatler yarışmanın başlangıcından itibaren sayılır. Tema provasındaki sayaç bu planın kısaltılmış hâlidir.

  1. 0 – 1. saat · Anla

    Temayı ve kuralları dinleyin, örnek parkuru izleyin. Sorularınızı o an sorun. Gelen malzemeyi sayıp listeleyin.

  2. 1 – 2. saat · Planla

    Herkes bir fikir söylesin, üç seçenekten birini seçin. Robotu kâğıda çizin: gövde, sensörlerin yeri, mekanizma.

  3. 2 – 8. saat · Yap

    Mekanik gövdeyi, elektronik kabloları, yazılım taslağı temaya uyarlamayı aynı anda yapar. 8. saatte robot yürümeli.

  4. 8 – 10. saat · İlk sürüş

    Motor yönleri, sensör eşikleri, servo açıları. Her denemeyi test defterine yazın.

  5. 10 – 14. saat · Dinlenme vardiyası

    İkişer kişi dönüşümlü dinlenir. Yorgun ekip hata yapar; vardiyayı planlayın.

  6. 14 – 20. saat · Görevi çöz

    Görevi baştan sona deneyin, en çok kaybettiren tek sorunu düzeltin, tekrar deneyin.

  7. 20 – 22. saat · Dondur

    Yeni özellik yok. Çalışan kodu kaydedin, kabloları bağlarla toplayın, yedek alın.

  8. 22 – 24. saat · Sun ve taşı

    Sunum provası yapın. Robotu pist alanına taşıyın: bitmeyen robot değerlendirilmez ama pist alanına taşınmak zorunda.

Diskalifiye riski

  • K'Nex'e yapıştırıcı, bant, silikon ya da makaron; parçalara kalıcı zarar.
  • 24 saat boyunca yarışma alanına yarışmacı ve görevliler dışında birinin girmesi; ekip diskalifiye edilebilir. Öğretmen ve veliler de alana girmez.
  • Koçun robota ya da öğrenciye müdahale etmesi, yönlendirmesi; uyarıya rağmen sürerse ekip diskalifiye edilir.
  • Başka robotlara ya da pistlere kasıtlı zarar vermek; izinsiz parkura girmek.
  • Centilmenliğe aykırı davranış. İtirazlar yalnız yazılı olarak kriz masasına yapılır.
