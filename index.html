Hazır kod modülleri

Önce dört testi ayrı ayrı çalıştırın. Sonra başlangıç taslağını açın: aynı parçalar tek dosyada, görev adımları durum makinesi olarak hazır. Yarışmada yalnızca görev adımlarını temaya göre değiştirirsiniz.

Başlangıç taslağı. Motor + 4'lü çizgi + HC-SR04 + servo, görev adımlarıyla.

// TYY BASLANGIC TASLAGI - Odemis BILSEM
// Pinler ORNEKTIR. Kendi baglantiniza gore degistirin.
#include <Servo.h>

// ---- PINLER ----
const byte SOL_PWM = 5, SOL_IN1 = 7, SOL_IN2 = 8;   // L298N sol motor
const byte SAG_PWM = 6, SAG_IN1 = 9, SAG_IN2 = 10;  // L298N sag motor
const byte TRIG = 11, ECHO = 12;                    // HC-SR04
const byte SERVO_PIN = 3;                           // Servo sinyal
const byte CIZGI[4] = {A0, A1, A2, A3};             // 4'lu sensor, soldan saga

// ---- AYARLAR (testte bulun) ----
const bool SIYAH_LOW = true;   // Siyahta sensor LOW veriyorsa true
const int  HIZ = 90;           // Temel hiz (0-255)
const int  YAKIN_CM = 10;      // Nesneye bu kadar yaklasinca dur
const int  SERVO_ACIK = 110, SERVO_KAPALI = 55;

Servo kol;

// ---- GOREV ADIMLARI: temaya gore burayi degistirin ----
enum Adim { BASLA, CIZGI_TAKIP, YAKLAS, AL, DONUS, BIRAK, BITTI };
Adim adim = BASLA;
unsigned long adimBaslangic = 0;

void adimaGec(Adim yeni) {
  adim = yeni;
  adimBaslangic = millis();
  Serial.print("Adim: ");
  Serial.println(yeni);
}
unsigned long adimSuresi() { return millis() - adimBaslangic; }

// ---- MOTOR ----
void motor(byte pwm, byte in1, byte in2, int hiz) {
  hiz = constrain(hiz, -255, 255);
  digitalWrite(in1, hiz >= 0 ? HIGH : LOW);
  digitalWrite(in2, hiz >= 0 ? LOW : HIGH);
  analogWrite(pwm, abs(hiz));
}
void sur(int sol, int sag) {
  motor(SOL_PWM, SOL_IN1, SOL_IN2, sol);
  motor(SAG_PWM, SAG_IN1, SAG_IN2, sag);
}
void dur() { sur(0, 0); }

// ---- MESAFE ----
long mesafeCm() {
  digitalWrite(TRIG, LOW);  delayMicroseconds(2);
  digitalWrite(TRIG, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG, LOW);
  unsigned long sure = pulseIn(ECHO, HIGH, 25000);
  return sure == 0 ? 0 : sure / 58;   // 0 = olcum yok
}

// ---- CIZGI ----
bool s[4];
int sonHata = 0;
bool cizgiVar(byte pin) {
  int d = digitalRead(pin);
  return SIYAH_LOW ? d == LOW : d == HIGH;
}
int sensorSay() {   // Kac goz cizgiyi goruyor? (0-4)
  int n = 0;
  for (byte i = 0; i < 4; i++) { s[i] = cizgiVar(CIZGI[i]); if (s[i]) n++; }
  return n;
}
void cizgiTakip(int hiz) {   // P kontrol, agirliklar -3 -1 +1 +3
  const int AGIRLIK[4] = {-3, -1, 1, 3};
  long toplam = 0; int n = 0;
  for (byte i = 0; i < 4; i++) if (s[i]) { toplam += AGIRLIK[i]; n++; }
  int hata = (n > 0) ? toplam / n : sonHata;
  sonHata = hata;
  int duzeltme = hata * 25;
  sur(hiz + duzeltme, hiz - duzeltme);
}

void setup() {
  Serial.begin(9600);
  pinMode(SOL_PWM, OUTPUT); pinMode(SOL_IN1, OUTPUT); pinMode(SOL_IN2, OUTPUT);
  pinMode(SAG_PWM, OUTPUT); pinMode(SAG_IN1, OUTPUT); pinMode(SAG_IN2, OUTPUT);
  pinMode(TRIG, OUTPUT); pinMode(ECHO, INPUT);
  for (byte i = 0; i < 4; i++) pinMode(CIZGI[i], INPUT);
  kol.attach(SERVO_PIN);
  kol.write(SERVO_ACIK);
  dur();
  adimaGec(BASLA);
}

void loop() {
  int gorenSensor = sensorSay();

  switch (adim) {
    case BASLA:                       // 2 sn bekle, sonra yola cik
      if (adimSuresi() > 2000) adimaGec(CIZGI_TAKIP);
      break;

    case CIZGI_TAKIP: {               // Cizgiyi izle, onde nesne varsa yaklas
      cizgiTakip(HIZ);
      long cm = mesafeCm();
      if (cm > 0 && cm < 25) adimaGec(YAKLAS);
      break;
    }

    case YAKLAS: {                    // Yavasca nesneye yaklas
      long cm = mesafeCm();
      if (cm == 0) dur();             // Olcum yoksa ilerleme
      else if (cm > YAKIN_CM) sur(60, 60);
      else { dur(); adimaGec(AL); }
      break;
    }

    case AL:                          // Kolu kapat, biraz geri cekil
      kol.write(SERVO_KAPALI);
      if (adimSuresi() > 600) sur(-60, -60);
      if (adimSuresi() > 1000) { dur(); adimaGec(DONUS); }
      break;

    case DONUS:                       // Yerinde don (sureyi olcun)
      sur(80, -80);
      if (adimSuresi() > 700) { dur(); adimaGec(BIRAK); }
      break;

    case BIRAK:                       // Hedef: 4 sensor de siyah (genis bant)
      if (gorenSensor == 4) {
        dur();
        kol.write(SERVO_ACIK);
        adimaGec(BITTI);
      } else {
        cizgiTakip(70);
      }
      break;

    case BITTI:
      dur();
      break;
  }
  delay(10);
}

Motor testi. Tekerlekler havadayken çalıştırın: ileri, dur, geri.

// L298N icin ornek Uno pinleri
const byte SOL_PWM = 5, SOL_IN1 = 7, SOL_IN2 = 8;
const byte SAG_PWM = 6, SAG_IN1 = 9, SAG_IN2 = 10;

void motor(byte pwm, byte in1, byte in2, int hiz) {
  hiz = constrain(hiz, -255, 255);
  digitalWrite(in1, hiz >= 0 ? HIGH : LOW);
  digitalWrite(in2, hiz >= 0 ? LOW : HIGH);
  analogWrite(pwm, abs(hiz));
}
void sur(int sol, int sag) {
  motor(SOL_PWM, SOL_IN1, SOL_IN2, sol);
  motor(SAG_PWM, SAG_IN1, SAG_IN2, sag);
}

void setup() {
  pinMode(SOL_PWM, OUTPUT); pinMode(SOL_IN1, OUTPUT); pinMode(SOL_IN2, OUTPUT);
  pinMode(SAG_PWM, OUTPUT); pinMode(SAG_IN1, OUTPUT); pinMode(SAG_IN2, OUTPUT);
}

void loop() {
  sur(90, 90);   delay(1000);  // Ileri. Bir teker geri donuyorsa kablolarini degistir.
  sur(0, 0);     delay(700);   // Dur
  sur(-80, -80); delay(700);   // Geri
  sur(0, 0);     delay(1500);
}

Çizgi sensörü testi. Seri Monitör'de dört gözün değerini görün; siyahta 0 mı 1 mi?

// 4'lu sensorun OUT1-OUT4 pinleri soldan saga A0-A3'e
const byte CIZGI[4] = {A0, A1, A2, A3};

void setup() {
  Serial.begin(9600);
  for (byte i = 0; i < 4; i++) pinMode(CIZGI[i], INPUT);
}

void loop() {
  for (byte i = 0; i < 4; i++) {
    Serial.print(digitalRead(CIZGI[i]));
    Serial.print(' ');
  }
  Serial.println();
  delay(100);
}
// Siyah cizgide 0 goruyorsan taslakta SIYAH_LOW = true kalsin.
// Kartin uzerindeki vidalarla (trimpot) hassasiyeti ayarla.

Mesafe testi. HC-SR04 ile santimetre ölçün. 0 gelirse ölçüm yok demektir.

const byte TRIG = 11, ECHO = 12;

long mesafeCm() {
  digitalWrite(TRIG, LOW);  delayMicroseconds(2);
  digitalWrite(TRIG, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG, LOW);
  unsigned long sure = pulseIn(ECHO, HIGH, 25000); // En fazla 25 ms bekle
  return sure == 0 ? 0 : sure / 58;
}

void setup() {
  Serial.begin(9600);
  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);
}

void loop() {
  long cm = mesafeCm();
  if (cm == 0) Serial.println("Olcum yok");
  else { Serial.print(cm); Serial.println(" cm"); }
  delay(150);
}

Servo testi. Kolun güvenli açık ve kapalı açısını bulun.

#include <Servo.h>
Servo kol;
const byte SERVO_PIN = 3;

void setup() {
  kol.attach(SERVO_PIN);
}

void loop() {
  kol.write(55);  delay(800);   // Kapali denemesi
  kol.write(90);  delay(800);   // Orta
  kol.write(110); delay(800);   // Acik denemesi
}
// Servo zorlaniyor/vizildiyorsa aciyi daralt.
// Bulduğun degerleri taslaktaki SERVO_ACIK ve SERVO_KAPALI'ya yaz.
Pinler örnektir. Yarışmada verilecek elektronik parçaların modeli kurallarda yazmıyor. Farklı bir sürücü ya da sensör çıkarsa yalnız ilgili modülü değiştirin.
Motoru Arduino'nun 5V pininden beslemeyin. Motorlar sürücüden ve pilden beslenir; tüm GND'ler ortak olmalı.
Her çalışan sürümü kaydedin. Örnek: tyy_v3_kiskac_calisiyor.ino. Bir şey bozulursa bir önceki sürüme dönersiniz.

Yarışma günü: 24 saat planı

Saatler yarışmanın başlangıcından itibaren sayılır. Tema provasındaki sayaç bu planın kısaltılmış hâlidir.

  1. 0 – 1. saat · Anla

    Temayı ve kuralları dinleyin, örnek parkuru izleyin. Sorularınızı o an sorun. Gelen malzemeyi sayıp listeleyin.

  2. 1 – 2. saat · Planla

    Herkes bir fikir söylesin, üç seçenekten birini seçin. Robotu kâğıda çizin: gövde, sensörlerin yeri, mekanizma.

  3. 2 – 8. saat · Yap

    Mekanik gövdeyi, elektronik kabloları, yazılım taslağı temaya uyarlamayı aynı anda yapar. 8. saatte robot yürümeli.

  4. 8 – 10. saat · İlk sürüş

    Motor yönleri, sensör eşikleri, servo açıları. Her denemeyi test defterine yazın.

  5. 10 – 14. saat · Dinlenme vardiyası

    İkişer kişi dönüşümlü dinlenir. Yorgun ekip hata yapar; vardiyayı planlayın.

  6. 14 – 20. saat · Görevi çöz

    Görevi baştan sona deneyin, en çok kaybettiren tek sorunu düzeltin, tekrar deneyin.

  7. 20 – 22. saat · Dondur

    Yeni özellik yok. Çalışan kodu kaydedin, kabloları bağlarla toplayın, yedek alın.

  8. 22 – 24. saat · Sun ve taşı

    Sunum provası yapın. Robotu pist alanına taşıyın: bitmeyen robot değerlendirilmez ama pist alanına taşınmak zorunda.
